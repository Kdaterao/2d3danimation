#version 330 core
out vec4 FragColor;
in vec2 TexCoord;
uniform sampler2D texture1;
uniform float uOpacity;
uniform float uColorScale;
uniform vec3 uTint;
void main() {
    vec4 colors = texture(texture1, TexCoord);

    // MyPaint / RGBM64 uses 15-bit range — scale RGB and alpha (was colors *= 2.0)
    FragColor = vec4(colors.rgb * uColorScale * uTint, colors.a * uColorScale) * uOpacity;

}
